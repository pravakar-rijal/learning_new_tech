    private String processPlaceholders(String text) {
        String result = text;
        if (text.contains("#{DATE")) {
            result = this.processDatePlaceholder(text);
        }

        if (result.contains("#{RD-")) {
            result = this.processRandomNumberPlaceholder(result);
        }

        if (result.contains("#{DB-")) {
            result = this.processDatabasePlaceholder(result);
        }

        if (result.contains("#{ST-")) {
            result = this.processStoredTextPlaceholder(result);
        }

        return result;
    }

    private String processDatePlaceholder(String text) {
        String result = text;

        int endIndex;
        for(int startIndex = text.indexOf("#{DATE"); startIndex != -1; startIndex = result.indexOf("#{DATE", endIndex)) {
            endIndex = result.indexOf("}", startIndex);
            String placeholder = result.substring(startIndex, endIndex + 1);
            String offsetStr = placeholder.substring(7, placeholder.length() - 1);
            int offset = Integer.parseInt(offsetStr);
            SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd");
            Date currentDate = new Date();
            long newDateMillis = currentDate.getTime() + (long)(offset * 24 * 60 * 60 * 1000);
            Date newDate = new Date(newDateMillis);
            String formattedDate = sdf.format(newDate);
            result = result.replace(placeholder, formattedDate);
        }

        return result;
    }

    private String processRandomNumberPlaceholder(String text) {
        String result = text;

        int endIndex;
        for(int startIndex = text.indexOf("#{RD-"); startIndex != -1; startIndex = result.indexOf("#{RD-", endIndex)) {
            endIndex = result.indexOf("}", startIndex);
            String placeholder = result.substring(startIndex, endIndex + 1);
            String digitsStr = placeholder.substring(5, placeholder.length() - 1);
            int digits = Integer.parseInt(digitsStr);
            int randomNumber;
            if (digits <= 0) {
                randomNumber = 0;
            } else {
                int lowerBound = (int)Math.pow((double)10.0F, (double)(digits - 1));
                int upperBound = (int)Math.pow((double)10.0F, (double)digits) - 1;
                Random random = new Random();
                randomNumber = random.nextInt(upperBound - lowerBound + 1) + lowerBound;
            }

            String randomNumberStr = String.valueOf(randomNumber);
            result = result.replace(placeholder, randomNumberStr);
        }

        return result;
    }

    private String processDatabasePlaceholder(String text) {
        String result = text;

        int endIndex;
        for(int startIndex = text.indexOf("#{DB-"); startIndex != -1; startIndex = result.indexOf("#{DB-", endIndex)) {
            endIndex = result.indexOf("}", startIndex);
            String placeholder = result.substring(startIndex, endIndex + 1);
            String key = placeholder.substring(5, placeholder.length() - 1);
            String value = this.getValueFromGlobalStoredResponse(key);
            result = result.replace(placeholder, value);
        }

        return result;
    }

    private String processStoredTextPlaceholder(String text) {
        String result = text;

        int endIndex;
        for(int startIndex = text.indexOf("#{ST-"); startIndex != -1; startIndex = result.indexOf("#{ST-", endIndex)) {
            endIndex = result.indexOf("}", startIndex);
            String placeholder = result.substring(startIndex, endIndex + 1);
            String key = placeholder.substring(5, placeholder.length() - 1);
            String value = this.getValueFromGlobalStoredResponse(key);
            result = result.replace(placeholder, value);
        }

        return result;
    }

    private String getValueFromGlobalStoredResponse(String key) {
        String globalStoredResponse = ((GlobalSharedData)GlobalSharedData.getInstance().get()).getGlobalStoredResponse();
        if (globalStoredResponse != null && !globalStoredResponse.isEmpty()) {
            JSONObject globalResponseJson = new JSONObject(globalStoredResponse);
            if (globalResponseJson.has(key)) {
                return globalResponseJson.getString(key);
            }
        }

        return "";
    }
