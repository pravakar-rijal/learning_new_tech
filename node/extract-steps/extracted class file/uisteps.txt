    @Test
    @And("User performs key action {string}")
    public void performKeyAction(String key) throws AbortScenario {
        if (!((GlobalSharedData)GlobalSharedData.getInstance().get()).shouldSkipScenario()) {
            try {
                Actions actions = new Actions(((UIManager)mgr.get()).driver);
                switch (key.toLowerCase()) {
                    case "space" -> actions.sendKeys(new CharSequence[]{Keys.SPACE}).perform();
                    case "alt" -> actions.keyDown(Keys.ALT).keyUp(Keys.ALT).perform();
                    case "left arrow" -> actions.sendKeys(new CharSequence[]{Keys.ARROW_LEFT}).perform();
                    case "right arrow" -> actions.sendKeys(new CharSequence[]{Keys.ARROW_RIGHT}).perform();
                    default -> throw new RuntimeException("Unsupported key action: " + key);
                }

                System.out.println("Performed key action: " + key);
            } catch (Exception e) {
                throw new RuntimeException("Error performing key action '" + key + "': " + e.getMessage(), e);
            }
        }
    }

    @When("User compares the image placed at {string} with the image placed at {string} and shows the difference in the image saved at {string} with color code {string}")
    public void compareImagesWithColorCode(String firstImagePath, String secondImagePath, String pathOfOutputFile, String colorCode) throws AbortScenario {
        if (!((GlobalSharedData)GlobalSharedData.getInstance().get()).shouldSkipScenario()) {
            Scalar highlightColor = this.parseColorCode(colorCode);
            ImageComparison.compareImagesHighlights(firstImagePath, secondImagePath, true, pathOfOutputFile, highlightColor);
        }
    }

    private Scalar parseColorCode(String colorCode) {
        try {
            String[] rgb = colorCode.split(",");
            if (rgb.length != 3) {
                throw new IllegalArgumentException("Invalid color code format. Expected format: R,G,B");
            } else {
                int r = Integer.parseInt(rgb[0].trim());
                int g = Integer.parseInt(rgb[1].trim());
                int b = Integer.parseInt(rgb[2].trim());
                return new Scalar((double)b, (double)g, (double)r, (double)0.0F);
            }
        } catch (NumberFormatException var6) {
            throw new IllegalArgumentException("Invalid color code values. Must be integers in the format: R,G,B");
        }
    }
