    public boolean isKeyPresent(JsonNode jsonNode, String targetKey) {
        if (jsonNode.isArray()) {
            for(JsonNode node : jsonNode) {
                if (this.isKeyPresent(node, targetKey)) {
                    return true;
                }
            }
        } else if (jsonNode.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> fields = jsonNode.fields();

            while(fields.hasNext()) {
                Map.Entry<String, JsonNode> entry = (Map.Entry)fields.next();
                if (((String)entry.getKey()).equals(targetKey)) {
                    return true;
                }

                if (this.isKeyPresent((JsonNode)entry.getValue(), targetKey)) {
                    return true;
                }
            }
        }

        return false;
    }

    private Map<String, String> parseExpectedValues(String expectedValuesString) {
        Map<String, String> expectedValues = new HashMap();
        String cleanedString = expectedValuesString.substring(1, expectedValuesString.length() - 1);
        String[] pairs = cleanedString.split(",");

        for(String pair : pairs) {
            String[] keyValue = pair.split(":");
            String key = keyValue[0].trim();
            String value = keyValue[1].trim();
            expectedValues.put(key, value);
        }

        return expectedValues;
    }

    private boolean validateDataType(String expectedPattern, String actualValue) {
        switch (expectedPattern) {
            case "#string":
                return APIUtils.inferType(actualValue).equals("String");
            case "#integer":
                return APIUtils.inferType(actualValue).equals("Integer");
            case "#double":
                return APIUtils.inferType(actualValue).equals("Double");
            case "#boolean":
                return APIUtils.inferType(actualValue).equals("Boolean");
            case "null":
                return APIUtils.inferType(actualValue).equals("null");
            case "#date":
                return actualValue.matches("\\d{4}-\\d{2}-\\d{2}");
            default:
                if (!expectedPattern.matches("#\\[\\d+-\\d+\\]")) {
                    this.assertions.fail("Unsupported pattern type: " + expectedPattern);
                    return false;
                } else {
                    String range = expectedPattern.substring(2, expectedPattern.length() - 1);
                    int lowerBound = Integer.parseInt(range.split("-")[0]);
                    int upperBound = Integer.parseInt(range.split("-")[1]);
                    int intValue = Integer.parseInt(actualValue);
                    return intValue >= lowerBound && intValue <= upperBound;
                }
        }
    }

    @Test
    @Then("User verifies all links are working fine")
    public void BrokenLinks() throws AbortScenario {
        if (!((GlobalSharedData)GlobalSharedData.getInstance().get()).shouldSkipScenario()) {
            this.checkAllLinks();
        }
    }

    public void checkAllLinks() {
        List<WebElement> links = ((UIManager)this.mgr.get()).driver.findElements(By.tagName("a"));
        int checkedLinksCount = 0;

        for(WebElement link : links) {
            String url = link.getAttribute("href");
            if (url != null && !url.isEmpty()) {
                try {
                    HttpURLConnection connection = (HttpURLConnection)(new URL(url)).openConnection();
                    connection.setRequestMethod("HEAD");
                    connection.connect();
                    int respCode = connection.getResponseCode();
                    if (respCode >= 400) {
                        System.out.println(url + " is a broken link");
                        this.assertions.assertTrue(false, "Broken link found: " + url, (UIManager)this.mgr.get());
                    } else {
                        System.out.println(url + " is a valid link");
                    }

                    ++checkedLinksCount;
                } catch (Exception var8) {
                    System.out.println("Error checking link: " + url);
                    this.assertions.assertTrue(false, "Error in connection: " + url, (UIManager)this.mgr.get());
                }
            }
        }

        System.out.println("Total links checked: " + checkedLinksCount);
    }

            LogUtil.log("Accessibility report has been written to " + filePath, LogLevel.INFO);
        }

    }

    private void createDirectoryIfNotExists(String filePath) {
        File directory = (new File(filePath)).getParentFile();
        if (!directory.exists()) {
            directory.mkdirs();
        }

    }

    private void writeReportToFile(String report, String filePath) throws IOException {
        try (FileWriter writer = new FileWriter(filePath)) {
            writer.write("----------------------Accessibility Report Starts");
            writer.write(report);
            writer.write("----------------------Accessibility Report Ends");
        } catch (IOException e) {
            e.printStackTrace();
        }

    }

    public static String buildCustomReport(Results axeResult) {
        int ruleCounter = 1;
        int nodeCounter = 1;
        StringBuilder reportBuilder = new StringBuilder();

        for(Rule rule : axeResult.getViolations()) {
            reportBuilder.append("_________________________________________________________________________________\n").append(ruleCounter).append(") ").append(rule.getDescription()).append(".").append("\n").append(rule.getHelpUrl()).append("\n");

            for(CheckedNode checkedNode : rule.getNodes()) {
                reportBuilder.append(ruleCounter).append(".").append(nodeCounter).append(") ").append(checkedNode.getTarget()).append("\n").append(checkedNode.getFailureSummary()).append("\n\n");
                ++nodeCounter;
            }

            ++ruleCounter;
            nodeCounter = 1;
        }

        return reportBuilder.toString();
    }

    private int convertTimeToSeconds(String time) {
        try {
            String[] timeParts = time.split(":");
            int seconds = 0;
            int multiplier = 1;

            for(int i = timeParts.length - 1; i >= 0; --i) {
                seconds += Integer.parseInt(timeParts[i]) * multiplier;
                multiplier *= 60;
            }

            return seconds;
        } catch (NumberFormatException e) {
            throw new RuntimeException("Invalid time format: " + time, e);
        }
    }
