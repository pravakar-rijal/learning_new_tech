                    JsonNode specialCookiesNode = mapper.readTree(((GlobalSharedData)sharedData.get()).getCookies());
                    ((ObjectNode)existingCookiesNode).setAll((ObjectNode)specialCookiesNode);
                }

                requestData.set(5, existingCookiesNode.toString());
                ApiRequest request = new ApiRequest();
                request.setEnvironment(StellarOptions.getInstance().getActiveEnvironment().toLowerCase());
                Map<String, String> replaceableParams = APIUtils.overrideRequestData(requestData, request.getEnvironment(), "requestdata");
                Map<String, String> props = APIUtils.convertMapKeysToLowerCase(APIUtils.overrideRequestData(requestData, request.getEnvironment(), "props"));
                requestData.set(1, APIUtils.replaceJsonPlaceholders((String)requestData.get(1), replaceableParams));
                requestData.set(2, APIUtils.replaceJsonPlaceholders((String)requestData.get(2), replaceableParams));
                requestData.set(3, APIUtils.replaceJsonPlaceholders((String)requestData.get(3), replaceableParams));
                iterativeData = IterativeUtils.populateCredentials(requestData, iterativeSheet);
                iterativeData.setRequestBody(IterativeUtils.getIterativeDataRowForJson(iterativeSheet, requestData));
                JSONObject queryParamsAndPathParams = new JSONObject((String)requestData.get(3));
                JSONObject queryParams = queryParamsAndPathParams.getJSONObject("QueryParams");
                JSONObject pathParams = queryParamsAndPathParams.getJSONObject("PathParams");

                for(int j = 0; j < iterativeData.getUsername().size(); ++j) {
                    for(int i = 0; i < rowCount; ++i) {
                        request = new ApiRequest();
                        request.setUsername(iterativeData.getUsernameatIndex(j));
                        request.setPassword(iterativeData.getPasswordatIndex(j));
                        request.setRequestData(requestData);
                        request.setRequestType(requestType);
                        request.setEndPoint(((APIManager)apiManager.get()).getConfigUtil().getEndpoint(endPoint));
                        request.setProps(props);
                        request.setRequestBody(IterativeUtils.replaceJsonKeys((String)requestData.get(1), (HashMap)iterativeData.getRequestBody(), i));
                        request.setRequestBody(request.getRequestBody().substring(1, request.getRequestBody().length() - 1).replace("\\n", "\n").replace("\\\"", "\""));
                        request.setHeaders((String)requestData.get(2));
                        request.setQueryParams(queryParams.toString());
                        request.setPathParams(pathParams.toString());
                        request.setCookies((String)requestData.get(5));
                        request.setServerName(server);
                        request.setResponse(((APIManager)apiManager.get()).serverRequestWithJson(request, false, ""));
                        this.requestList.insertRequestAtLast(request);
                        ((GlobalSharedData)sharedData.get()).setLatestStoredResponse(request.getResponse());
                        APIUtils.PrintRequestLogs(request);
                        String maskedResponse = APIUtils.maskingValuesXml(((GlobalSharedData)sharedData.get()).getLatestStoredResponse().body().asString(), APIUtils.extractSensitiveDataKeys((String)request.getRequestData().get(4), request.getEnvironment()));
                        LogUtil.log("Response Of Request          ==================== " + maskedResponse, LogLevel.valueOf(ConstantKeys.logType));
                        APIUtils.PrintResponseLogs(((GlobalSharedData)sharedData.get()).getLatestStoredResponse());
                    }
                }

            }
        } catch (CommonException e) {
            e.printStackTrace();
            throw e;
        }
    }
